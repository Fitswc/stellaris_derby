namespace = uma_season

country_event = {
	id = uma_season.0
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_global_flag = uma_global_game_setting_gamesystem_season_enable
	}

	immediate = {
		every_planet_within_border = {
			limit = {
				is_colony = yes
			}
			carrier_event = {
				id = uma_season.1
				days = 12
			}
		}
		country_event = {
			id = uma_season.9
			days = 185
		}
	}
}

country_event = {
	id = uma_season.2
	is_triggered_only = yes
	hide_window = yes

	trigger = {

	}

	immediate = {
		every_owned_planet = {
			limit = {
				is_colony = yes
				OR = {
					has_deposit = d_uma_season_spring
					has_deposit = d_uma_season_summer
					has_deposit = d_uma_season_autumn
					has_deposit = d_uma_season_winter
				}
			}
			carrier_event = {
				id = uma_season.3
			}
		}
	}
}


#Changing season on planet(Colonies only)
carrier_event = {
	id = uma_season.3
	is_triggered_only = yes
	hide_window = yes

	immediate = {
		switch = {
			trigger = has_deposit

			d_uma_season_spring = {
				add_deposit = d_uma_season_summer
				remove_deposit = d_uma_season_spring
			}

			d_uma_season_summer = {
				add_deposit = d_uma_season_autumn
				remove_deposit = d_uma_season_summer
			}

			d_uma_season_autumn = {
				add_deposit = d_uma_season_winter
				remove_deposit = d_uma_season_autumn
			}

			d_uma_season_winter = {
				add_deposit = d_uma_season_spring
				remove_deposit = d_uma_season_winter
			}
		}
		last_added_deposit = {
			set_deposit_flag = uma_keep_deposit_on_terraform
			set_deposit_flag = keep_deposit_on_terraform
		}
	}

	after = {
		carrier_event = {
			id = uma_season.4
			days = 5
		}
	}
}
