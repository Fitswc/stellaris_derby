building_uma_basic_racing_course = {
	base_buildtime = 200
	category = unity

	building_sets = {
		uma_racing
		unity
		urban
		government
	}


	potential = {
		exists = owner
		owner = {
			OR = {
				has_authority = auth_trainingcenter
				#is_uma_fallen_empire = yes
			}
		}
		NOT = { has_modifier = slave_colony }
	}

	destroy_trigger = {
		exists = owner
		OR = {
			owner = {
				NOR = {
					has_authority = auth_trainingcenter
					#is_uma_fallen_empire = yes
				}
			}
			has_modifier = slave_colony
		}
	}

	resources = {
		category = planet_buildings
#		cost = {
#			minerals = 360
#		}
		inline_script = {
			script = buildings/nomadic_cost_switcher
			REGULAR_RESOURCE = minerals
			NOMADIC_RESOURCE = alloys
			COST = 360
			NOMADIC_COST_MULT = 0.2
		}
		upkeep = {
			energy = 1
		}
	}

	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				OR = {
					has_authority = auth_trainingcenter
					#is_uma_fallen_empire = yes
				}
			}
		}
		modifier = {
			job_uma_racing_umamusume_add = 100

			job_uma_racing_ura_members_add = 50


		}
	}

	triggered_desc = {
		trigger = {
			exists = owner
			owner = {
				OR = {
					has_authority = auth_trainingcenter
					#is_uma_fallen_empire = yes
				}
			}
		}
		text = uma_job_uma_racing_umamusume_effect_desc
	}

	prerequisites = {
		"tech_uma_race_tradition"
	}

	upgrades = {
		building_uma_modern_racing_course
	}

	show_tech_unlock_if = {
		OR = {
			has_authority = auth_trainingcenter
			#is_uma_fallen_empire = yes
		}
	}

}

building_uma_dormitary = {
	base_buildtime = 240
	category = amenity

	building_sets = {
		government
		urban
		entertainment
		resort
		uma_urban
		uma_entertainment
	}

	potential = {
		OR = {
			NOT = { is_planet_class = pc_habitat }
			AND = {
				exists = owner
				owner = {
					is_void_dweller_empire = yes
				}
			}
		}
		NOR = {
			has_modifier = slave_colony
			has_modifier = penal_colony
		}
		exists = owner
		owner = {
			is_regular_empire = yes
			has_authority = auth_trainingcenter
		}
		NOT = { has_building = building_low_tech_capital }
	}

	destroy_trigger = {
		exists = owner
		OR = {
			AND = {
				owner = {
					is_ai = yes
				}
			}
			owner = {
				OR = {
					is_regular_empire = no
					NOT = {
						has_authority = auth_trainingcenter
					}
				}
			}
			has_modifier = slave_colony
		}
	}


	planet_modifier = {
		planet_housing_add = 3500
		planet_amenities_add = 5600
		pop_environment_tolerance = 0.02
	}

	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				has_active_tradition = tr_domination_imperious_architecture
			}
		}
		modifier = {
			planet_housing_add = 100
		}
	}

	resources = {
		category = planet_buildings
		cost = {
			alloys = 600
		}
		upkeep = {
			energy = 4
		}
	}

	upgrades = {
		"building_uma_paradise"
	}
}
