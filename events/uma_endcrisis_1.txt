system_event = {
	id = uma_ec_1.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		any_system_megastructure = { has_megastructure_flag = uma_ec1_dyson_sphere }
	}

	immediate = {
		random_system_megastructure = {
			limit = { has_megastructure_flag = uma_ec1_dyson_sphere }
			prev = {
				spawn_megastructure = {
					type = uma_ec1_dyson_sphere_ruined
					coords_from = prev
					init_effect = {
						if = {
							limit = { exists = fromfrom }
							set_megastructure_flag = uma_ruined_ec1_dyson_sphere@fromfrom
						}
					}
				}
			}
			remove_megastructure = this
		}
	}
}

fleet_event = {
	id = uma_ec_1.3
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		queue_actions = {
			repeat = {
				while = {
					id = uma_ec_1.3.e1

					NOT = {
						any_system = {
							has_star_flag = uma_ec1_side_defend_system
							count_fleet_in_system = {
								count < 2
							}
						}
					}
				}
				find_closest_system = {
					trigger = {
						id = uma_ec_1.3.t1
						count_fleet_in_system = {
							count < 2
						}
					}
					found_system = {
						move_to = this
						find_random_planet = {
							trigger = {
								id = uma_ec_1.3.t2

								is_star = yes
							}
							found_planet = {
								move_to = this
							}
						}
					}
					failed = {
						wait = {
							duration = 10
						}
					}
				}
			}
		}
	}
}

fleet_event = {
	id = uma_ec_1.4
	hide_window = yes
	is_triggered_only = yes

	trigger = {

		owner = {
			NOT = {
				is_same_value = event_target:uma_ec1_country
				is_hostile_to = event_target:uma_ec1_country
			}
		}

		From = {
			has_star_flag = uma_ec1_main_system
		}

	}

	immediate = {
		while = {
			count = 2

			random_galaxy_fleet = {
				limit = {
					has_fleet_flag = uma_ec1_def_fleet
				}
				clear_fleet_actions = this
				queue_actions = {
					find_closest_system = {
						trigger = {
							id = uma_ec_1.4

							has_star_flag = uma_ec1_main_system
						}
						found_system = {
							move_to = this
						}
					}
				}
			}
		}
	}
}
