uma_ec3_create_puma_fleets = {
	create_fleet = {
		name = "NAME_UMA_EC3_PUMA_FLEET"
		effect = {
			set_fleet_flag = uma_ec3_puma_front_fleet
			while = {
				count = 30
				create_ship = {
					name = random
					design = NAME_UMA_EC3_FRONT_SQUARD_BB
					graphical_culture = mammalian_01
				}
			}

			while = {
				count = 45
				create_ship = {
					name = random
					design = NAME_UMA_EC3_FRONT_SQUARD_CV
					graphical_culture = mammalian_01
				}
			}

			set_fleet_stance = aggressive
			set_aggro_range_measure_from = self
			set_aggro_range = 500
			set_fleet_bombardment_stance = uma_bs_soc
		}
	}
}

uma_ec3_create_puma_front_behaviour = {
	queue_actions = {
		repeat = {
			find_closest_system = {
				trigger = {
					id = uma_ec3_puma_front_fleet_trigger.1
					NOT = {
						is_owned_by = event_target:uma_ec3_uma_from_outer_world_country
					}
				}
				found_system = {
					move_to = this
					wait = 150
				}
			}
		}
	}
}
