country_event = {
	id = uma_zephyr.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		Root = {
			has_authority = auth_trainingcenter
			NOT = {
				has_country_flag = uma_zephyr_events_meet_zephyr
			}
		}
		From = {
			is_surveyed = {
				who = Root
				status = yes
			}

		}
	}

	immediate = {
		Root = {
			set_country_flag = uma_zephyr_events_meet_zephyr
		}
		From = {
			log = "UMA_ZEPHY_SURVEY_HAPPENED_AT_SURVEY_WHICH_IN_[From]"
			# Scope check
		}
		From = { #system
			create_country = {
				name = uma_zephyr_country_name_unknown
				type = global_event
				ignore_initial_colony_error = yes
				authority = random
				flag = random
				species = random
				name_list = random #?
				origin = origin_default
				effect = {
					set_graphical_culture = Root
					save_global_event_target_as = uma_zephyr_fleet_country
					create_ship_design = {
						design = NAME_UMA_SHIP_ZEPHYR
					}

					Prev = {
						create_fleet = {
							name = NAME_UMA_SHIP_ZEPHYR
							effect = {
								save_global_event_target_as = uma_zephyr_fleet
								create_ship = {
									name = NAME_UMA_SHIP_ZEPHYR
									design = last_created_design
									graphical_culture = Root
								}

								set_location = Prev.star
							}
						}
					}
				}
			}
		}

		country_event = {
			id = uma_zephyr.2
		}
	}
}

	id = uma_zephyr.20
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		This.fleet.owner = {
			has_authority = auth_trainingcenter

			NOT = {
				has_country_flag = uma_zephyr_events_tiyanki_happened
			}
			has_country_flag = uma_zephyr_events_chain_begin
		}

		From = {
			OR = {
				has_star_flag = tiyanki_graveyard_system
			}
		}
	}

	immediate = {
		From = {
			create_fleet = {
				name = NAME_UMA_SHIP_ZEPHYR
				effect = {
					create_ship_design = {
						design = NAME_UMA_SHIP_ZEPHYR
					}
					save_global_event_target_as = uma_zephyr_fleet
					create_ship = {
						name = NAME_UMA_SHIP_ZEPHYR
						design = last_created_design
						graphical_culture = event_target:uma_zephyr_fleet_country
						effect = {
							set_disable_at_health = 0.01
						}
					}

					set_location = This
					set_cloaking_active = yes
				}
			}
			event_target:uma_zephyr_fleet_country = {
				set_faction_hostility = {
					target = Root.fleet.owner
					set_friendly = yes
				}
			}
		}
		This.fleet.owner = {
			set_country_flag = uma_zephyr_events_tiyanki_happened
