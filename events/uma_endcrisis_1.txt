namespace = uma_ec_1

#When crisis Begin
country_event = {
	id = uma_ec_1.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		NOR = {
			has_global_flag = uma_ec1_timemachine_happened
			has_global_flag = uma_ec_invasion_ec1_end
			has_global_flag = uma_ec_invasion_end
		}
	}

	immediate = {
		endgame_telemetry = uma_end_crisis_1
		set_global_flag = uma_ec1_timemachine_happened
		set_global_flag = uma_ec_invasion_happened

		log = "UMA ENDCRISIS1 WILL HAPPENED AT [GetDate]"

		create_species = {
			name = "NAME_ec1_threegodness_species"
			class = umamusume
			portrait = umamusume
			traits = random
			immortal = yes
			name_list = UMAname
			effect = { save_global_event_target_as = uma_ec1_umamusume_species }
		}
		create_country = {
			name = "NAME_UMA_EC1_country"
			type = uma_ec3_empire
			species = event_target:uma_ec1_umamusume_species
			name_list = "UMAname"
			flag = {
				icon = {
					category = "special"
					file = "unknown.dds"
				}
				background = {
					category = "backgrounds"
					file = "flag_BG_39.dds"
				}
				colors = {
					"light_blue"
					"moss_green"
					"null"
					"null"
				}
			}

			effect = {
				create_ship_design = {
					design = NAME_starbase_uma_ec1
				}
				add_ship_design = last_created_design

				set_country_flag = uma_ec1_country
				save_global_event_target_as = uma_ec1_country

				set_graphical_culture = humanoid_01
			}
		}

		random_system = {
			limit = {
				is_star_class = sc_pulsar
			}
			if = {
				limit = {
					exists = starbase
				}
				starbase = {
					fleet = {
						destroy_fleet = this
					}
				}
			}
			spawn_megastructure = {
				type = uma_ec1_dyson_sphere
				planet = solar_system.star
				coords_from = this
				orbit_angle = random
				orbit_distance = 5
				owner = event_target:uma_ec1_country
				graphical_culture = mammalian_01
				random_pos = yes
				init_effect =  {
					set_megastructure_flag = uma_ec1_dyson_sphere
				}
			}

			create_starbase = {
				size = starbase_uma_ec1
				module = shipyard
				module = detection_array
				effect = {
					while = {
						limit = {
							fleet = {
								fleet_power >= 12000
							}
						}
						create_ship = {
							design = NAME_UMA_EC1_ION_CANNON
						}
					}
				}
			}

			every_neighbor_system = {
				limit = {
					distance = {
						source = Prev
						max_jumps = 1
					}
				}
				if = {
					limit = {
						exists = starbase
					}
					starbase = {
						fleet = {
							destroy_fleet = this
						}
					}
				}
				create_starbase = {
					size = starbase_uma_ec1
					effect = {
						create_ship = {
							design = NAME_UMA_EC1_ION_CANNON
						}
					}
				}
				set_star_flag = uma_ec1_side_defend_system
			}
		}
	}
}


##On Win
system_event = {
	id = uma_ec_1.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		any_system_megastructure = { has_megastructure_flag = uma_ec1_dyson_sphere }
	}

	immediate = {
		random_system_megastructure = {
			limit = { has_megastructure_flag = uma_ec1_dyson_sphere }
			prev = {
				spawn_megastructure = {
					type = uma_ec1_dyson_sphere_ruined
					coords_from = prev
					init_effect = {
						if = {
							limit = { exists = fromfrom }
							set_megastructure_flag = uma_ruined_ec1_dyson_sphere@fromfrom
						}
					}
				}
			}
			remove_megastructure = this
		}
	}
}

fleet_event = {
	id = uma_ec_1.3
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		queue_actions = {
			repeat = {
				while = {
					id = uma_ec_1.3.e1

					any_system = {
						has_star_flag = uma_ec1_side_defend_system
						count_fleet_in_system = {
							count < 2
						}
					}
				}
				find_closest_system = {
					trigger = {
						id = uma_ec_1.3.t1
						count_fleet_in_system = {
							count < 2
						}
					}
					found_system = {
						move_to = this
						find_random_planet = {
							trigger = {
								id = uma_ec_1.3.t2

								is_star = yes
							}
							found_planet = {
								move_to = this
							}
						}
					}
					failed = {
						wait = {
							duration = 10
						}
					}
				}
			}
		}
	}
}
