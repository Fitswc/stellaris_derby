namespace = uma_bg

planet_event = {
	id = uma_bg.1
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		owner = {
			has_authority = auth_trainingcenter
		}
	}

	immediate = {
		if = {
			limit = {
				OR = {
					has_active_building = building_commercial_zone
					has_active_building = building_commercial_megaplex
				}
				NOT = {
					has_planet_flag = uma_has_trader_exist
				}
			}
			set_planet_flag = uma_has_trader_exist
			add_modifier = {
				modifier = uma_add_treater
				clear_on_owner_change = yes
			}
		}
		if = {
			limit = {
				NOT = {
					OR = {
						has_active_building = building_commercial_zone
						has_active_building = building_commercial_megaplex
					}
					has_planet_flag = uma_has_trader_exist
				}
			}
			remove_planet_flag = uma_has_trader_exist
			remove_modifier = uma_add_treater
		}

	}
}

					trait = uma_commander_trait_sp_seiun_sky_2
				}
			}
		}
	}
}

country_event = {
	id = uma_bg.33
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		FromFrom = {
			any_owned_ship = {
				has_component = UMA_REALM_STICKER
			}
			is_in_combat = yes
		}
	}

	immediate = {
		random_list = {
			6 = {
				FromFromFrom = {
					export_trigger_value_to_variable = {
						trigger = count_owned_ship
						variable = uma_bg33_count_total_ship_num
					}
					multiply_variable = {
						value = 0.6
						which = uma_bg33_count_total_ship_num
					}

					while = {
						count = uma_bg33_count_total_ship_num

						random_owned_ship = {
							reduce_shield = 100000 #All reduce
						}
					}
				}
			}

			4 = {
				Root = {
					add_resource = {
						influence = 0
					}
				}
			}
		}
	}

	after = {
		FromFromFrom = {
			clear_variable = uma_bg33_count_total_ship_num
		}

		country_event = {
			id = uma_bg.33
			days = 20
		}
	}
}
