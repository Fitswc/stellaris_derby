uma_ap2_system_variable_handler = {
	optimize_memory
	$METHOD$_variable = {
		which = uma_ap2_$AREA_TYPE$_rate
		value = 1
	}
}


uma_ap2_factor_summation = {
	change_variable = {
		which = uma_ap2_factor_summation
		value = owner.uma_ap2_factor_charm_rate
