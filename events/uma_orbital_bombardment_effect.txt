planet_event = {
	id = probuma.1
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		has_orbital_bombardment_stance = uma_bs_lightoff
		is_colony = yes
	}
	immediate = {
		if = {
			limit = {
				uma_is_ringworld = yes
			}
			set_planet_flag = cb_b_r
		}

		else_if = {
			limit = {
				uma_is_habitat = yes
			}
			set_planet_flag = cb_b_h
		}
		else = {
			set_planet_flag = cb_b
		}

	}

}

planet_event = {
	id = probuma.2
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		OR = {
			has_planet_flag = cb_b
			has_planet_flag = cb_b_h
			has_planet_flag = cb_b_r
		}
	}

	immediate = {
		if = {
			limit = {
				has_planet_flag = cb_b_r
			}
			change_pc = pc_shattered_ring_habitable
			remove_planet_flag = cb_b_r
		}
		else_if = {
			limit = {
				has_planet_flag = cb_b_h
			}
			remove_planet = yes
			remove_planet_flag = cb_b_h
		}
		else = {
			change_pc = pc_shattered
			remove_planet_flag = cb_b
		}

	}
}




planet_event = {
	id = probuma.3
	is_triggered_only = yes
	hide_window = yes


	immediate = {
		export_trigger_value_to_variable = {
			trigger = count_owned_pop
			parameters = {
				limit = {

				}
			}
			variable = uma_pk_planet_pop_will_not_be_transform
		}

		divide_variable = {
			which = uma_pk_planet_pop_will_not_be_transform
			value = 2
		}


		every_owned_pop = {
			modify_species = {
				species = From.owner_main_species
				add_trait = trait_uma_new_bottle_but_old_wine
				change_scoped_species = yes
				#					portrait = From.owner_main_species
				effect = {
					##
				}
			}
		}

		while = {
			count = uma_pk_planet_pop_will_not_be_transform

			random_owned_pop = {
				kill_pop = yes
			}
		}



		set_owner = From.owner
	}
}
