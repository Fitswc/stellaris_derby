namespace = uma_zephyr


country_event = {
	id = uma_zephyr.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		Root = {
			has_authority = auth_trainingcenter
		}
		From = {
			is_surveyed = {
				who = Root
				status = yes
			}

		}
	}

	immediate = {
		From = {
			log = "UMA_ZEPHY_SURVEY_HAPPENED_AT_SURVEY_WHICH_IN_[From]"
			# Scope check
		}
		From = { #system
			create_country = {
				name = uma_zephyr_country_name_unknown
				type = global_event
				ignore_initial_colony_error = yes
				authority = random
				flag = random
				species = random
				name_list = random #?
				origin = origin_default
				effect = {
					set_graphical_culture = Root
					save_global_event_target_as = uma_zephyr_fleet_country
					create_ship_design = {
						design = NAME_UMA_SHIP_ZEPHYR
					}

					Prev = {
						create_fleet = {
							name = NAME_UMA_SHIP_ZEPHYR
							effect = {
								save_global_event_target_as = uma_zephyr_fleet
								create_ship = {
									name = NAME_UMA_SHIP_ZEPHYR
									design = last_created_design
									graphical_culture = Root
								}

								set_location = {
									target = Prev.star
									direction = out_system
									distance = {
										max = 20
										min = 15
									}
									angle = random
								}
							}
						}

					}
				}
			}




		}
	}
}

ship_event = {
	id = uma_zephyr.10
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		This.fleet.owner = {
			has_authority = auth_trainingcenter
			NOT = {
				has_country_flag = uma_zephyr_events_observer_happened
			}
		}
	}

	immediate = {
		This.fleet.owner = {
			country_event = {
				id = uma_zephyr.11
			}
		}
		From = {
			create_country = {
				name = uma_zephyr_country_name
				type = global_event
				ignore_initial_colony_error = yes
				authority = random
				flag = random
				species = random
				name_list = random #?
				origin = origin_default
				effect = {
					set_graphical_culture = Root.fleet.owner
					save_global_event_target_as = uma_zephyr_fleet_country
					create_ship_design = {
						design = NAME_UMA_SHIP_ZEPHYR
					}

					Prev = {
						create_fleet = {
							name = NAME_UMA_SHIP_ZEPHYR
							effect = {
								save_global_event_target_as = uma_zephyr_fleet
								create_ship = {
									name = NAME_UMA_SHIP_ZEPHYR
									design = last_created_design
									graphical_culture = Root.fleet.owner
								}

								set_location = {
									target = Prev
									distance = {
										max = 20
										min = 15
									}
									angle = random
								}
							}
						}
					}
				}
			}
		}
	}
}
