	trigger = {
		FROM = {
			has_authority = auth_trainingcenter
		}
		any_planet_within_border = {
			check_variable = {
				which = uma_count_bad_corp_building_country

				has_authority = auth_trainingcenter
			}

		base = -10

			multiply = this.uma_count_bad_corp_building_country

		}
	}
