uma_factor_speed_add_modifier = {
	if = {
		limit = {
			uma_check_factor_rate = {
				TYPE = speed
				VAL = 1
			}
		}

	}
	else_if = {
		limit = {
			uma_check_factor_rate = {
				TYPE = speed
				VAL = 2
			}
		}
	}
	else_if = {
		limit = {
			uma_check_factor_rate = {
				TYPE = speed

			}
		}
	}
	else_if = {
		limit = {
			uma_check_factor_rate = {
				TYPE = speed

			}
		}
	}
	else_if = {
		limit = {

			}
		}
	}
	else_if = {
		limit = {

			}
		}
	}
	else_if = {

		}

}
