# Frontend Backgrounds
#
# Defines the selectable animated backgrounds shown on the main menu.
# Each entry maps a script tag to an entity name defined in a .asset file.
#
# -----------------------------------------------------------------------
# MODDING GUIDE
# -----------------------------------------------------------------------
#
# To add a new selectable background, create a file in this folder
# (load order determines dropdown order) and add an entry:
#
#   my_mod_background = {
#       entity = "my_mod_frontend_root_entity"
#   }
#
# The display name is looked up as a loc key: <tag>_FRONTEND_NAME
# For the example above that is:  my_mod_background_FRONTEND_NAME
#
# The entity named in "entity" must be defined in a .asset file that is
# loaded by the engine. For a custom mod background, ship your .asset file
# alongside this script file.
#
# To change which background is shown when the player has not yet made a
# selection ("Latest Release" mode), set "default = yes" on one entry.
# Only one entry should carry this flag. Clausewitz last-loaded-wins
# applies: a file loaded after this one can override the default by
# setting "default = yes" on any entry, including those defined here.
#
# Entries are listed in release order. The right arrow in the settings
# spinner cycles forward through this list.
#
# -----------------------------------------------------------------------

uma = {
	entity = "uma_frontend_root_entity"
	default = yes
}
