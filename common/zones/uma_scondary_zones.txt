zone_uma_housing_ring_world = {
	icon = GFX_district_specialization_uma_housing
	base_buildtime = @zone_buildtime
	potential = {
		hidden_trigger = { exists = owner }
		owner = {
			has_authority = auth_trainingcenter
		}
	}

	unlock = {
		hidden_trigger = { exists = owner }
		owner = {
			has_authority = auth_trainingcenter
		}
	}

	resources = {
		category = planet_zones
		cost = {
			minerals = @zone_cost
		}
	}

	triggered_district_planet_modifier = {
		planet_housing_add = 300
	}

	max_buildings = 3

	include = {
		all
	}

	excluded_building_sets = {
		generator
		mining
		farming
		knights
		betharian
		generator_automation
		mining_automation
		farming_automation
		urban_automation_arcology
		urban_automation

		uma_farming
		uma_generator
		uma_mining
	}

	swap_type = district_uma_ring_housing
	swap_type_weight = 5

	zone_sets = {
		umaring_world
	}

	planet_modifier = {
		zone_building_slots_add = 3
	}

	triggered_desc = {
		trigger = {
			always = yes
		}
		text = zone_urban_triggered_desc
	}

	ai_priority = 2
	ai_weight_coefficient = 0.5
}
