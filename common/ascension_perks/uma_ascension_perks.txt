
###新飞升
##全身全灵（领域飞升一阶段）
ap_uma_zenshizenren = {
	on_enabled = {
		custom_tooltip = "uma_will_enable_zenshizenren_ap"
		#		custom_tooltip = ap_uma_zenshizenren_ap_two_tt
		hidden_effect = {
			country_event = {
				id = uma_ap.2
			}
		}
		###小草的对话
		if = {
			limit = {
				has_country_flag = uma_have_communicated_with_grass_wonder
			}
			country_event = {
				id = uma_special_leader.23
			}
		}
	}
	possible = {
		custom_tooltip = {
			fail_text = "requires_ascension_perks_1"
			num_ascension_perks > 0
		}
		#		has_technology = tech_uma_terriroty_in_use
		custom_tooltip = {
			fail_text = "uma_must_be_toresen_auth"
			has_authority = "auth_trainingcenter"
		}
	}
	potential = {
		host_has_dlc = "Utopia"
		NOT = {
			has_ascension_perk = ap_uma_zenshizenren
			has_authority = "auth_machine_intelligence"
		}
		is_mechanical_empire = no
		is_cyborg_empire = no
	}
	ai_weight = {
		factor = 300
		modifier = {
			factor = 0.8
			NOR = {
				has_ethic = ethic_spiritualist
				has_ethic = ethic_fanatic_spiritualist
			}
		}
		modifier = {
			factor = 0
			num_owned_planets < 6
		}
	}
}

ap_uma_fly_me_to_the_star = {
	on_enabled = {
		custom_tooltip = "uma_will_enable_keisun_ap"
		#		custom_tooltip = ap_uma_fly_me_to_the_star_ap_two_tt
		hidden_effect = {
			country_event = {
				id = uma_ap.22
			}
			###小草的对话
			if = {
				limit = {
					has_country_flag = uma_have_communicated_with_grass_wonder
				}
				country_event = {
					id = uma_special_leader.22
				}
			}
		}
	}
	possible = {
		custom_tooltip = {
			fail_text = "requires_ascension_perks_1"
			num_ascension_perks > 0
		}
		# custom_tooltip = {
		# 	fail_text = "uma_requires_technology_uma_complex_keisun_test"
		# 	has_technology = tech_uma_complex_keisun_test
		# }
		#		has_technology = tech_uma_complex_keisun_test
		custom_tooltip = {
			fail_text = "uma_must_be_toresen_auth"
			has_authority = "auth_trainingcenter"
		}
		custom_tooltip = {
			fail_text = origin_uma_after_factor_perk_already
			NOT = {
				has_origin = origin_uma_after_factor_perk
			}
		}
	}
	modifier = {
		UMA_species_trait_points_add = 3
		country_government_civic_points_add = 1
	}
	potential = {
		host_has_dlc = "Utopia"
		NOT = {
			has_ascension_perk = ap_uma_fly_me_to_the_star
			has_authority = "auth_machine_intelligence"
		}
		is_mechanical_empire = no
		is_cyborg_empire = no
	}
	ai_weight = {
		factor = 300
		modifier = {
			factor = 0.8
			NOR = {
				has_ethic = ethic_fanatic_materialist
				has_ethic = ethic_materialist
			}
		}
		modifier = {
			factor = 0
			num_owned_planets < 6
		}
	}
}
