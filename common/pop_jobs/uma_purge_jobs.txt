		}
		produces = {
			trigger = {

				}
			}
			food = 0.5
		}
		produces = {
			trigger = {
				owner = {

				}
			}

			society_research = 0.3
			engineering_research = 0.3
		}

		upkeep = {
			trigger = {
				is_robotic_species = yes
			}
			energy = 3
		}

		upkeep = {
			trigger = {
				owner = {
					NOT = {
						has_policy_flag = policy_uma_fuel_convertion_produce_food
					}
				}

			}

		}
