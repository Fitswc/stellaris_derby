namespace = uma_bg

planet_event = {
	id = uma_bg.1
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		owner = {
			has_authority = auth_trainingcenter
		}
	}

	immediate = {
		if = {
			limit = {
				OR = {
					has_active_building = building_commercial_zone
					has_active_building = building_commercial_megaplex
				}
				NOT = {
					has_planet_flag = uma_has_trader_exist
				}
			}
			set_planet_flag = uma_has_trader_exist
			add_modifier = {
				modifier = uma_add_treater
				clear_on_owner_change = yes
			}
		}
		if = {
			limit = {
				NOT = {
					OR = {
						has_active_building = building_commercial_zone
						has_active_building = building_commercial_megaplex
					}
					has_planet_flag = uma_has_trader_exist
				}
			}
			remove_planet_flag = uma_has_trader_exist
			remove_modifier = uma_add_treater
		}

	}
}

	hide_window = yes
	is_triggered_only = yes

	trigger = {
		exists = owner
		owner = { has_technology = tech_uma_explore_4 }
		NOT = { owner = { has_technology = tech_uma_explore_5 } }
		NOT = { has_planet_flag = uma_has_uesd_colony_buff }
	}

	immediate = {
		set_planet_flag = uma_has_uesd_colony_buff
		add_modifier = {
			modifier = uma_prosperity_colony_1
			days = 3650
		}
	}
}

	id = uma_bg.18
	hide_window = yes
	is_triggered_only = yes


	immediate = {
		every_country = {
			limit = {
				OR = {
					has_authority = auth_corporate
					has_authority = auth_cyber_creed_corporate

					has_authority = auth_cyber_corporate_individualist
					has_authority = auth_synth_corporate_physical
					has_authority = auth_synth_corporate_virtual
				}
			}

			type = cb_uma_business_war
			who = event_target:uma_it_is_corp_and_we_can_make_it_pay_tax
			days = -1
		}
	}
}
