ap_uma_become_the_crisis = {
	potential = {
		has_nemesis = yes
		is_uma_country = yes
		NOR = {
			is_player_crisis = yes
			has_ascension_perk = ap_defender_of_the_galaxy
		}
		is_wilderness_empire = no
	}

	possible = {
		custom_tooltip = {
			fail_text = ap_uma_become_the_crisis_not_in_multiplayer
			NOT = {
				has_global_flag = uma_ap_uma_become_the_crisis_actived
			}
		}
		#Disable For Debug
		custom_tooltip = {
			fail_text = "requires_independence"
			is_subject = no
		}
		custom_tooltip = {
			fail_text = "must_not_be_custodian"
			is_galactic_custodian = no
		}
		custom_tooltip = {
			fail_text = "must_not_be_emperor"
			is_galactic_emperor = no
		}
		custom_tooltip = {
			fail_text = "must_finish_shroud_forged_situation"
			OR = {
				NOT = { has_origin = origin_shroud_forged }
				NOT = {
					any_situation = {
						is_situation_type = situation_shroud_forged
					}
				}
			}
		}
		custom_tooltip = {
			fail_text = "locked_by_shroud_forged_situation"
			OR = {
				NOT = { has_origin = origin_shroud_forged }
				NOT = { has_country_flag = finish_shroud_forged_liberation_flag }
			}
		}

		custom_tooltip = {
			fail_text = "requires_ascension_perks_3"
			num_ascension_perks > 2
		}

		custom_tooltip = {
			fail_text = "uma_become_the_crisis_must_had_ap1"
			AND = {
				has_ascension_perk = ap_uma_fly_me_to_the_star
				has_ascension_perk = ap_uma_zenshizenren
			}
		}
	}

	on_enabled = {
		hidden_effect = {
			country_event = { id = uma_ap.6999 } # Timeline Become the crisis
			hidden_effect = {
				set_variable = {
					which = uma_ap_bc_menace_produced
					value = 0
				}
			}
			set_global_flag = uma_ap_uma_become_the_crisis_actived
			set_country_flag = is_player_crisis
		}
#		activate_crisis_progression = uma_realm_break_to_realworld_path
		country_event = {
			id = uma_ap.300
		}
	}

	ai_weight = {
		factor = 0
	}
}
