uma_ap_system_factor_speed_minus = {
    potential = {
        always = yes
    }
    allow = {
#        always = yes
        check_variable = {
            which = uma_ap2_factor_speed_rate
            value > 0
        }
    }
    effect = {
        custom_tooltip = uma_ap_system_factor_speed_t_tt
        hidden_effect = {
            uma_ap2_system_variable_handler = {
                METHOD = subtract
                AREA_TYPE = factor_speed
            }
            uma_ap2_factor_summation = yes
        }
    }
}
