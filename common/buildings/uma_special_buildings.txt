building_uma_nice_nature_no_restaurant = {
	base_buildtime = 300
	category = unity
#	base_cap_amount = 1
	planet_limit = {
		base = 1
	}

	building_sets = {
		uma_urban
		uma_racing
		uma_research
		urban
		government
	}
	
	potential = {
		exists = owner
		owner = {
			OR = {
				has_authority = auth_trainingcenter
				#is_uma_fallen_empire = yes
			}
		}
		NOT = { has_modifier = slave_colony }
	}

	destroy_trigger = {
		exists = owner
		OR = {
			owner = {
				NOR = {
					has_authority = auth_trainingcenter
					#is_uma_fallen_empire = yes
				}
			}
			has_modifier = slave_colony
		}
	}

	allow = {
		has_upgraded_capital = yes
	}

	resources = {
		category = planet_buildings
		cost = {
			minerals = 300
		}
		upkeep = {
			energy = 2
		}
	}

	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				OR = {
					has_authority = auth_trainingcenter
					#is_uma_fallen_empire = yes
				}
			}
		}
		modifier = {
			pop_happiness = 0.2
			country_consumer_goods_produces_mult = 0.33
			planet_jobs_trade_produces_mult = 0.33
		}
	}

	prerequisites = {
		"tech_uma_nice_nature_no_restaurant"
	}

	show_tech_unlock_if = {
		OR = {
			has_authority = auth_trainingcenter
			#is_uma_fallen_empire = yes
		}
	}
}

building_realm_planetary_shield_generator = {
	base_buildtime = @b3_time
#	base_cap_amount = 1
	planet_limit = {
		base = 1
	}
	position_priority = 100
	can_be_ruined = no
	category = army

	building_sets = {
		government
		fortress
		uma_fortress
	}

	potential = {
		exists = owner
		owner = {
			OR = {
				has_tradition = uma_realm_field_large_job_to_use
				has_ascension_perk = ap_uma_mixture_ap_2
			}
			OR = {
				has_authority = auth_trainingcenter
			}
		}
	}


	allow = {
		hidden_trigger = { owner = { is_ai = no } }
		if = {
			limit = {
				exists = orbital_defence
			}
			orbital_defence = {
				exists = starbase
				starbase = {
					NOR = {
						has_starbase_building = ring_orbital_shield_generator
						is_starbase_building_building = ring_orbital_shield_generator
					}
				}
			}
		}
	}

	destroy_trigger = {
		exists = owner
		owner = {
			is_ai = yes
			NOT = {
				has_authority = auth_trainingcenter
			}
		}
	}

	planet_modifier = {
		planet_orbital_bombardment_damage = -0.85
		army_defense_health_mult = 0.4
	}

	resources = {
		category = planet_buildings
		cost = {
			minerals = 1000
			uma_crystal_carrot = 300
		}
		upkeep = {
			uma_crystal_carrot = 10
			energy = 90
		}
	}

}
