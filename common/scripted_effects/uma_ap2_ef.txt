	}

uma_ap_bc_beyond_l3_resource_control_counting_monthly_calculator = {
	optimize_memory
	divide_variable = {
		which = uma_ap_bc_count_playable_country
		value = uma_ap_bc_count_playable_country_initial
	}

	if = {
		limit = {
			check_variable = {
				which = uma_ap_bc_count_playable_country
				value >= 0.9
			}
		}
		set_country_flag = uma_ap_bc_ec_resource_penatly_P4
	}
	else_if = {
		limit = {
			check_variable = {
				which = uma_ap_bc_count_playable_country
				value >= 0.7
			}
			check_variable = {
				which = uma_ap_bc_count_playable_country
				value < 0.9
			}
		}
		set_country_flag = uma_ap_bc_ec_resource_penatly_P3
	}
	else_if = {
		limit = {
			check_variable = {
				which = uma_ap_bc_count_playable_country
				value >= 0.4
			}
			check_variable = {
				which = uma_ap_bc_count_playable_country
				value < 0.7
			}
		}
		set_country_flag = uma_ap_bc_ec_resource_penatly_P2
	}
	else_if = {
		limit = {
			check_variable = {
				which = uma_ap_bc_count_playable_country
				value >= 0.2
			}
			check_variable = {
				which = uma_ap_bc_count_playable_country
				value < 0.4
			}
		}
		set_country_flag = uma_ap_bc_ec_resource_penatly_P1
	}
	else_if = {
		limit = {
			check_variable = {
				which = uma_ap_bc_count_playable_country
				value < 0.2
			}
		}
		set_country_flag = uma_ap_bc_ec_resource_penatly_P0
	}
}
