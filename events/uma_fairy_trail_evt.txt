country_event = {
	id = uma_f_t.2
	title = uma_f_t.2.title
	desc = uma_f_t.2.desc
	picture = GFX_evt_bone_building
	is_triggered_only = yes

	trigger = {
		any_system_within_border = {
			distance = {
				source = root.solar_system
				min_jumps = 1
				max_jumps = 15
			}
			any_system_planet = {
				uma_is_normal_habitat_planet = yes
				is_under_colonization = no
			}
		}
	}

	immediate = {
		set_country_flag = uma_fairy_trail_intro_event_happened
		random_system_within_border = {
			limit = {
				distance = {
					source = root.solar_system
					min_jumps = 1
					max_jumps = 15
				}
				any_system_planet = { 
					OR = {
						uma_is_normal_habitat_planet = yes
					}
					is_under_colonization = no
				}
			}
			save_event_target_as = uma_fairy_trail_system_here
			log = "UMA_FAIRY_TRAIL_SYSTEM_FOUND,NAME = [event_target:uma_fairy_trail_system_here.GetName]"
			random_system_planet = {
				limit = {
					uma_is_normal_habitat_planet = yes
					is_under_colonization = no
				}
				save_global_event_target_as = uma_fairy_trail_planet_here
				log = "UMA_FAIRY_TRAIL_SYSTEM_FOUND,NAME = [event_target:uma_fairy_trail_planet_here.GetName]"
			}
		}
	}

	option = {
		name = uma_f_t.2.a
		hidden_effect = {
			event_target:uma_fairy_trail_planet_here = {
				create_archaeological_site = uma_fairy_trail_arch
			}
		}
	}

	option = {
		name = uma_f_t.2.b
		hidden_effect = {
			add_resource = {
				influence = 100
			}
		}
		default_hide_option = yes
	}
}
