#Yearly Pulse
event = {
	id = uma_ec_3.8065
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_crises_allowed = yes
		NOT = {
			has_global_flag = uma_ec3_outeruma_happened
		}

		OR = {
			AND = { # Super early start prereqs
				end_game_years_passed >= 0
				any_playable_country = {
					has_authority = auth_trainingcenter
				}
				OR = {
					AND = { #Both crisis summoning resolutions active
						is_active_resolution = 	resolution_galacticstudies_extradimensional_experimentation
						is_active_resolution = resolution_galactic_threats_committee
					}
				}
			}
			AND = { # Early start
				end_game_years_passed >= 23
				any_playable_country = {
					has_authority = auth_trainingcenter
				}
				OR = {
					default_endgame_early_start_triggers = yes
					any_country = {
						is_country_type = default
						OR = {
							has_technology = tech_jump_drive_1
							has_technology = tech_psi_jump_drive_1
						}
					}
					is_active_resolution = resolution_galacticstudies_extradimensional_experimentation
					is_active_resolution = resolution_galactic_threats_committee
				}
			}
			end_game_years_passed >= 48 #Regular Start
		}
		OR = {
			NOT = { has_global_flag = galactic_crisis_happened }
			AND = {
				allowed_crisis_type = all
				NOR = {
					has_global_flag = galactic_crisis_recently_fired #because some might take a while to arrive, we don't want two at once
					any_country = {
						is_crisis_faction = yes
					}
				}
			}
		}
	}

	immediate = {
		random_country = {
			limit = {
				is_country_type = global_event
			}
			##
			random_list = {
				10 = {
					modifier = {
						factor = 10000000
						any_playable_country = {
							has_authority = auth_trainingcenter
							is_ai = no
						}
					}
					country_event = {
						id = uma_ec_3.1
						days = 2
					}
				}
				5 = {
					#NTH, to paradox crisis
					modifier = {
						subtract = 5
						any_playable_country = {
							has_authority = auth_trainingcenter
							is_ai = no
						}
					}
				}
			}
		}
	}
}




country_event = {
	id = uma_ec_3.1
	hide_window = yes
	fire_only_once = yes
	is_triggered_only = yes

	immediate = {
		set_update_modifiers_batch = begin
		endgame_telemetry = uma_end_crisis_3
		set_global_flag = uma_ec3_outeruma_happened


		random_system = {
			limit = {
				is_star_class = sc_black_hole
			}

			set_star_flag = uma_ec3_first_system
			save_global_event_target_as = uma_ec3_first_system

			random_system_planet = {
				create_species = {
					name = "NAME_goldship_species"
					class = umamusume
					portrait = umamusume
					traits = random
					immortal = yes
					name_list = UMAname
					effect = { save_global_event_target_as = uma_ec3_umamusume_species }
				}

				create_country = {
					name = "NAME_UMA_EC3_country"
					type = uma_ec3_empire
					species = event_target:uma_ec3_umamusume_species
					name_list = "UMAname"
					flag = {
						icon = {
							category = "uiu_flags"
							file = "uiu_flags02.dds"
						}
						background = {
							category = "backgrounds"
							file = "circle.dds"
						}
						colors = {
							"indigo"
							"blue"
							"null"
							"null"
						}
					}

					effect = {

						set_country_flag = uma_ec3_country
						save_global_event_target_as = uma_ec3_country

						set_graphical_culture = mammalian_01
						create_ship_design = { design = "NAME_UMA_EC3_STARBASE" }
						add_ship_design = last_created_design
						create_ship_design = { design = "NAME_Marauder_Starbase" }
						add_ship_design = last_created_design
					}
				}

				event_target:uma_ec3_country = {
					create_fleet = {
						name = NAME_UMA_EC3_MAIN_BASE
						effect = {
							set_owner = event_target:uma_ec3_country
							create_ship = {
								name = random
								design = "NAME_UMA_EC3_BASE"
								effect = {
									set_ship_flag = uma_ec3_main_base
								}
							}
							set_location = {
								target = PREVPREV
								distance = 45
								angle = random
							}

							save_global_event_target_as = uma_ec3_main_base
							set_fleet_flag = uma_ec3_main_base

							#开局200日后解锁
							set_event_locked = yes

							#第一批攻击/防守舰队生成
							fleet_event = {
								id = uma_ec_3.2
								days = 50
							}
							#防守
							fleet_event = {
								id = uma_ec_3.3
								days = 100
							}
						}
					}
				}
			}

			if = {
				limit = { exists = starbase }
				starbase = { fleet = { destroy_fleet = this } }
			}
			create_starbase = {
				size = starbase_marauder
				owner = event_target:uma_ec3_country
			}
		}

		every_playable_country = {
			capital_star = {
				create_fleet = {
					name = NAME_UMA_EC3_SOC_BASE
					effect = {
						set_owner = event_target:uma_ec3_country
						create_ship = {
							name = random
							design = "NAME_UMA_EC3_STARBASE"
							effect = {
								set_ship_flag = uma_ec3_soc_fleet
							}
						}
						set_location = {
							target = Prev
							distance = 6
							angle = random
						}

						save_global_event_target_as = uma_ec3_soc_fleet
						set_fleet_flag = uma_ec3_soc_fleet

						#达到目标自动摧毁
						set_event_locked = yes
					}
				}
			}
		}

		set_update_modifiers_batch = end
	}
}

fleet_event = {
	id = uma_ec_3.5
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_fleet_flag = uma_ec3_puma_front_fleet
	}

	immediate = {
		queue_actions = {
			repeat = {
				#5跳内舰队力量过小->折跃
				effect = {

				}

				find_closest_system = {
					trigger = {
						id = uma_ec_3.5.t1
						OR = {
							AND = {
								exists = starbase
								starbase = { owner = { is_hostile = root.owner } }
								#NOR = {
								#	#高难度下不再被星基吸引
								#	has_global_flag = wg_crisis_hard
								#	has_global_flag = wg_crisis_insane
								#}
							}
						}
						ROOT = { can_access_system = prev }

					}
					found_system = {
						effect = {
							id = uma_ec_3.5.e1
							set_variable = { which = uf_fleet_id value = root.fleet_id }
						}
						move_to = this

						repeat = {
							while = {
								id = uma_ec_3.5.t2

								NOR = {
									star = {
										is_star_class = sc_black_hole
									}
								}

							}
							find_closest_planet = {
								trigger = {
									id = uma_ec_3.5.t3
									is_star = yes
									NOT = {
										is_star_class = sc_black_hole
									}
								}
								found_planet = {
									orbit_planet = this
									effect = {
										id = uma_ec_3.5.e2
										planet_event = {
											id = uma_ec_3.6
											days = 200
										}
									}
								}
								failed = {
									effect = {
										id = uma_ec_3.5.e3
										create_fleet = {
											effect = {
												set_owner = event_target:uma_ec3_country
												create_ship = {
													name = random
													design = "NAME_UMA_EC3_STARBASE"
													effect = {
														set_ship_flag = uma_ec3_soc_fleet_base
													}
												}
												set_location = {
													target = Prev
													distance = 10
													angle = random
												}

												save_global_event_target_as = uma_ec3_soc_fleet_base
												set_fleet_flag = uma_ec3_soc_fleet_base
											}
										}
									}
								}
							}
							move_to = this
							wait = {duration = 10 random = 10}
						}
						wait = { duration = 10 random = 10 }
					}

					failed = {
						find_random_fleet = {
							trigger = {
								id = uma_ec_3.5.t4
								owner = {
									is_hostile = Root.owner
								}
							}
							found_fleet = {
								move_to = this
							}
						}
					}
				}
			}
		}
	}
}
