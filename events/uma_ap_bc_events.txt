country_event = {
	id = uma_ap.404
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		uma_is_uma_ap_bc_actived = yes
		has_country_flag = uma_ap.403.happened
	}

	immediate = {
		add_resource = {
			uma_menace = -800
		}
		export_resource_stockpile_to_variable = {
			resource = uma_menace
			variable = uma_ap.404.uma_menace.stock
		}
		multiply_variable = {
			which = uma_ap_404_uma_menace_stock
			value = -1
		}
		add_variable = {
			which = uma_ap_404_uma_menace_stock
			value = 16000
		}
		divide_variable = {
			which = uma_ap_404_uma_menace_stock
			value = 16000
		}
		reset_event_chain_counter = {
			event_chain = uma_ap_bc_main_evt_chain
			counter = uma_ap_bc_main_evt_chain_crisis_level_reached_discounting
		}
		add_event_chain_counter = {
			event_chain = uma_ap_bc_main_evt_chain
			counter = uma_ap_bc_main_evt_chain_crisis_level_reached_discounting
			amount = uma_ap_404_uma_menace_stock
		}
	}
}

country_event = {
	id = uma_ap.405
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		NOR = {
			has_global_flag = uma_ec_invasion_ec4_end
			has_global_flag = uma_ec_invasion_end
		}
	}

	immediate = {
		endgame_telemetry = uma_end_crisis_4
		set_global_flag = uma_ec_invasion_happened

		log = "UMA ENDCRISIS_AP HAPPENED AT [GetDate]"

		set_update_modifiers_batch = begin

		export_trigger_value_to_variable = {
			trigger = num_empires
			variable = uma_num_of_regular_empire
		}
		export_trigger_value_to_variable = {
			trigger = num_fallen_empires
			variable = uma_num_of_fallen_empire
		}
		change_variable = {
			which = uma_num_of_regular_empire
			value = uma_num_of_fallen_empire
		}

		export_trigger_value_to_variable = {
			trigger = num_ai_empires_setting
			variable = uma_num_of_regular_empire
		}
		export_trigger_value_to_variable = {
			trigger = num_fallen_empires_setting
			variable = uma_num_of_fallen_empire
		}
		change_variable = {
			which = uma_num_of_regular_empire
			value = uma_num_of_fallen_empire
		}
		## Get the number of available empire in the area

		every_country = {
			limit = {
				is_ai = yes
				OR = {
					is_country_type = default
					is_country_type = fallen_empire
					is_country_type = uma_fallen_empire
				}
			}
			set_country_flag = uma_ap_bc_will_be_transform
		}

		while = {
			limit = {
				any_country = {
					has_country_flag = uma_ap_bc_will_be_transform
				}
			}
			random_country = {
				limit = {
					has_country_flag = uma_ap_bc_will_be_transform
				}
				country_event = {
					id = uma_ap.406
					days = 2
				}
			}
		}
		set_update_modifiers_batch = end
	}

	after = {
		every_country = {
			limit = {
				has_country_flag = uma_ap_bc_ec_country_tred
			}
			country_event = {
				id = uma_ap.407
				days = 20
				random = 10
			}
		}
	}
}

fleet_event = {
	id = uma_ap.408
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		queue_actions = {
			repeat = {
				find_closest_system = {
					trigger = {
						id = uma_ap.408.t1
						OR = {
							AND = {
								exists = starbase
								starbase = { owner = { is_hostile = root.owner } }
							}

							any_galaxy_fleet = {
								owner = {
									is_guardian_country = no
								}
							}
						}

						ROOT = { can_access_system = prev }

						OR = {
							AND = {
								count_fleet_in_system = {
									limit = {
										owner = {
											has_country_flag = uma_ap_bc_ec_country_tred
										}
									}
									count <= 2
								}
								is_capital_system = no
							}

							AND = {
								is_capital_system = yes
							}
						}
					}
					found_system = {
						move_to = this

						repeat = {
							#if has colonies
							while = {
								id = uma_ap.408.t2

								NOT = {
									any_system_planet = {
										is_colony = yes
										is_under_colonization = no

										is_owned_by = event_target:uma_ap_bc_target_country
									}
								}
							}
							find_closest_planet = {
								trigger = {
									id = uma_ap.408.t3

									is_colony = yes
									is_under_colonization = no

									is_owned_by = event_target:uma_ap_bc_target_country
								}
								found_planet = {
									orbit_planet = this
									effect = {
										id = uma_ap.408.e1

										#Bombard the planet
										Root = {
											set_fleet_flag = uma_ap.408.bombingplanet
										}

										planet_event = {
											id = uma_ap.409
											days = 150
											scopes = {
												from = Root
											}
										}
									}
								}
							}
							move_to = this
							wait = { duration = 10 random = 10 }
						}
						wait = { duration = 10 random = 10 }
					}
					failed = {
						find_random_fleet = {
							trigger = {
								id = uma_ec_3.5.t4
								owner = {
									is_hostile = Root.owner
								}
							}
							found_fleet = {
								move_to = this
							}
						}
					}
				}
			}
		}
	}
}
