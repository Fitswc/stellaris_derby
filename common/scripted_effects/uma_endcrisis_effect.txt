

uma_ec3_create_puma_fleets = {
	optimize_memory
	event_target:uma_ec3_country = {
		create_leader = {
			class = commander
			species = event_target:uma_ec3_umamusume_species
			name = random
			skill = 5
			randomize_traits = yes
			hide_age = yes
		}
	}


	create_fleet = {
		name = NAME_UMA_EC3_SOC_Corvette_2
		 settings = {
            can_upgrade = no
            uses_naval_capacity = no
            can_change_composition = no
            can_disband = no
            spawn_debris = no
            ai_ignore_strength = yes
        }
		effect = {
			set_owner = event_target:uma_ec3_country
			set_fleet_flag = uma_ec3_puma_front_fleet
			while = {
				count = 30
				create_ship = {
					name = random
					design = NAME_UMA_EC3_FRONT_SQUARD_BB
					graphical_culture = mammalian_01
				}
			}
			assign_leader = last_created_leader

			while = {
				count = 60
				create_ship = {
					name = random
					design = NAME_UMA_EC3_FRONT_SQUARD_CV
					graphical_culture = mammalian_01
				}
			}

			while = {
				count = 80
				create_ship = {
					name = random
					design = NAME_UMA_EC3_SOC_Corvette_2
					graphical_culture = mammalian_01
				}
			}


			while = {
				count = 5
				create_ship = {
					name = random
					design = NAME_UMA_EC3_SOC_Titan_2
					graphical_culture = mammalian_01
				}
			}


			set_location = {
				target = Prev
				distance = 5
				angle = random
			}


			set_fleet_stance = aggressive
			set_aggro_range_measure_from = self
			set_aggro_range = 500
			set_fleet_bombardment_stance = uma_bs_soc

		}
	}

	last_created_fleet = {
		fleet_event = {
			id = uma_ec_3.5
		}

		if = {
			limit = {
				has_global_flag = uma_global_game_setting_end_game_crisis_diff_easy
			}
			add_modifier = {
				modifier = uma_endcrisis_ec3_fleet_buff_easy
			}
		}
		if = {
			limit = {
				has_global_flag = uma_global_game_setting_end_game_crisis_diff_hard
			}
			add_modifier = {
				modifier = uma_endcrisis_ec3_fleet_buff_hard
			}
		}
	}
}

uma_ec3_create_puma_soc_at_capitial = {
	optimize_memory
	create_fleet = {
		settings = {
			is_boss = yes
			ai_ignore_strength = yes
		}
		name = NAME_UMA_EC3_SOC_BASE
		effect = {
			set_owner = event_target:uma_ec3_country
			create_ship = {
				name = NAME_UMA_EC3_SOC_BASE
				design = "NAME_UMA_EC3_STARBASE"
				effect = {
					set_ship_flag = uma_ec3_soc_fleet
				}
			}
			set_location = {
				target = Prev
				distance = 6
				angle = random
			}

			save_global_event_target_as = uma_ec3_soc_fleet
			set_fleet_flag = uma_ec3_soc_fleet

			#达到目标自动摧毁
			set_event_locked = yes


			queue_actions = {
				find_closest_planet = {
					trigger = {
						id = uma_ec_3.1.e1

						is_star = yes
					}

					found_planet = {
						orbit_planet = this
					}
				}
			}
		}
	}
}

uma_ec3_puma_create_species_and_country = {
	optimize_memory
	create_species = {
		name = "NAME_goldship_species"
		class = umamusume
		portrait = umamusume
		traits = random
		immortal = yes
		name_list = UMAname
		effect = { save_global_event_target_as = uma_ec3_umamusume_species }
	}

	create_country = {
		name = "NAME_UMA_EC3_country"
		type = uma_ec3_empire
		species = event_target:uma_ec3_umamusume_species
		name_list = "UMAname"
		flag = {
			icon = {
				category = "uiu_flags"
				file = "uiu_flags02.dds"
			}
			background = {
				category = "backgrounds"
				file = "circle.dds"
			}
			colors = {
				"indigo"
				"blue"
				"null"
				"null"
			}
		}

		effect = {

			set_country_flag = uma_ec3_country
			save_global_event_target_as = uma_ec3_country

			set_graphical_culture = mammalian_01
			create_ship_design = { design = "NAME_UMA_EC3_STARBASE" }
			add_ship_design = last_created_design
			create_ship_design = { design = "NAME_Marauder_Starbase" }
			add_ship_design = last_created_design
		}
	}
}
