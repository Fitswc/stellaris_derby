	icon = "district_uma_dormitory_ring"

	zone_slots = {
		slot_city_government
		slot_city_01
		slot_city_02
	}

	inline_script = {
		script = districts/BPV_district_slots2
		GOVERNMENT = slot_city_government
		SLOT1 = slot_city_01
		SLOT2 = slot_city_02
	}

	show_on_uncolonized = {
		exists = from
		from = { is_regular_empire = yes }
		uses_district_set = uma_ring_world
	}

		exists = owner
		owner = { is_regular_empire = yes }
		uses_district_set = uma_ring_world
	}

		district_rw_nexus
		district_rw_hive
	}

		category = planet_districts

			minerals = 300
			uma_crystal_carrot = 30
		}
		upkeep = {
			energy = 5
			uma_crystal_carrot = 1
		}

		planet_housing_add = 1000
		job_clerk_add = 600
	}

	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { is_uma_country = yes }
		}
		job_uma_dormitory_administrator_add = 100
	}

	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { is_gestalt = yes }
		}
		modifier = {
			job_maintenance_drone_add = 900
		}

			}
		}
		modifier = {
			planet_housing_add = 3000
		}
	}

	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				has_technology = tech_housing_2
				NOT = { has_valid_civic = civic_agrarian_idyll }
			}
		}
		modifier = {
			planet_housing_add = 3000
		}
	}

	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				has_technology = tech_housing_2
				has_ethic = ethic_gestalt_consciousness
			}
		}
		modifier = {
			job_maintenance_drone_add = 300
		}
	}

	triggered_planet_modifier = {

			exists = owner

			}
