operation_uma_pop_looting_and_transform = {
	target = none
	categories = { op_cat_uma_pop_looting_and_transform op_cat_government }
	picture = GFX_evt_uma_zf_glx_rob
	desc = {
		trigger = { has_nemesis = yes }
		text = operation_uma_pop_looting_and_transform_desc
	}
	desc = {
		trigger = { has_nemesis = no }
		text = operation_uma_pop_looting_and_transform_no_dlc_desc
	}
	stages = 2

	resources = {
		category = operations
		cost = {
			alloys = 5
		}
		upkeep = {
			energy = 4
		}
	}

	spy_power_cost = 15
	potential = {
		owner = {
			has_origin = origin_uma_after_factor_perk
		}
		target = {
			is_country_type = default
			is_fallen_empire = no
		}

	}
	allow = {
		custom_tooltip = {
			is_running_espionage_operation = no
			fail_text = operation_one_at_a_time
		}
	}
	stage = {
		difficulty = 2
		icon = GFX_espionage_chapter_icon_surveillance
		event = uma_espionage.1
	}

	stage = {
		difficulty = 2
		icon = GFX_espionage_chapter_icon_motion
		event = uma_espionage.2
	}

	on_roll_failed = {
		standard_espionage_operation_on_roll_failed = {
			RANDOM_EVENTS = operation_random_events_low_stakes
		}
	}
	on_create = {
		target = {
			set_timed_country_flag = {
				flag = espionage_operation_targeted_by@root.owner #Used in random Spy Network events
				days = @operationTargetedByTimer
			}
		}
	}
}
