event = {
	id = uma_origin.1
    hide_window = yes

    is_triggered_only = yes

    immediate = {

		every_playable_country = {
			if = {
				limit = {
					OR = {
						has_origin = origin_uma_below_threegodness
						has_origin = origin_uma_candleoff
						has_origin = origin_uma_business_MAN
						has_origin = origin_uma_after_factor_perk
					}
				}
				log = UMA_ORIGINS_LAUNCH
				country_event = {
					id = uma_origin.2
					days = 6 ##From Better UMA By ATR
				}
				set_country_flag = custom_start_screen
			}
			else_if = {
				limit = {
					is_species_class = umamusume
					NOT = {
						OR = {
							has_origin = origin_uma_below_threegodness
							has_origin = origin_uma_candleoff
							has_origin = origin_uma_business_MAN
							has_origin = origin_uma_after_factor_perk
						}
					}
				}
				log = UMA_ZERO_LAUNCH
				country_event = {
					id = uma_origin.4
					days = 6 ##From Better UMA By ATR
				}
				set_country_flag = custom_start_screen
			}
		}
		every_playable_country = {
			limit = {
				is_ai = no
				has_authority = auth_trainingcenter
			}
			save_global_event_target_as = uma_player_umamusume_empire
		}
    }
}
