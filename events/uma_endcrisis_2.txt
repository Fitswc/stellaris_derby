fleet_event = {
	id = uma_ec_2.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_fleet_flag = uma_endcrisis_ec2_attack_fleet
	}

	immediate = {
		queue_actions = {
			repeat = {
				find_closest_system = {
					trigger = {
						id = uma_ec_2.2.t1
						OR = {
							AND = {
								if = {
									limit = {
										owner = {
											is_ai = yes
										}
									}
									is_capital_system = no
									exists = starbase
									starbase = { owner = { is_hostile = root.owner } }
								}
								else_if = {
									limit = {
										owner = {
											is_ai = no
										}
									}
									exists = starbase
									starbase = { owner = { is_hostile = root.owner } }
								}
							}
						}
						ROOT = { can_access_system = prev }
						NOT = {
							has_star_flag = uma_ec_2.2.e1.our_fleet_is_local_here
						}

					}
					found_system = {
						effect = {
							id = uma_ec_2.2.e1

							set_star_flag = uma_ec_2.2.e1.our_fleet_is_local_here
						}
						move_to = this

						repeat = {
							while = {
								id = uma_ec_2.2.t2

								NOT = {
									exists = starbase
									starbase = {
										is_owned_by = event_target:uma_fallen_empire
									}

								}
							}
							find_closest_planet = {
								trigger = {
									id = uma_ec_2.2.t3
									is_primary_star = yes
									NOT = {
										exists = starbase
										starbase = {
											owner = {
												is_same_value = event_target:uma_fallen_empire
											}
										}
									}
								}
								found_planet = {
									orbit_planet = this
									effect = {
										id = uma_ec_2.2.e2

										planet_event = {
											id = uma_ec_2.3
											days = 8
										}
									}
								}
							}
							move_to = this
							wait = { duration = 10 random = 5 }
						}
						wait = { duration = 10 random = 5 }
					}

					failed = {
						find_random_fleet = {
							trigger = {
								id = uma_ec_2.2.t4
								owner = {
									is_hostile = Root.owner
								}
							}
							found_fleet = {
								move_to = this
							}
						}
					}
				}
			}
		}
	}
}

country_event = {
	id = uma_ec_2.10
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		log = "DECREASED THE COUNTER CLASS EC.10"
		event_target:global_event_country = {
			set_variable = {
				which = uma_ec2_num_fallen_empires_middle
				value = uma_ec2_num_fallen_empires_current_be_occupied
			}
			divide_variable = {
				which = uma_ec2_num_fallen_empires_middle
				value = uma_ec2_num_fallen_empires_total
			}

			if = {
				limit = {
					check_variable = {
						which = uma_ec2_num_fallen_empires_middle
						value <=0.3
					}
				}

				add_event_chain_counter = {
					event_chain = uma_ec_2_main_evt_chain
					counter = uma_ec2_fallen_empire_class
					amount = -1
				}
				event_target:uma_fallen_empire = {
					remove_modifier = uma_fallen_empire_crisis_2_modifier_class_1_hull
					remove_modifier = uma_fallen_empire_crisis_2_modifier_class_1_speed
					remove_modifier = uma_fallen_empire_crisis_2_modifier_class_2_hull
					remove_modifier = uma_fallen_empire_crisis_2_modifier_class_2_speed
					remove_modifier = uma_fallen_empire_crisis_2_modifier_class_3_hull
					remove_modifier = uma_fallen_empire_crisis_2_modifier_class_4

					add_modifier = {
						modifier = uma_fallen_empire_crisis_2_modifier_class_4
					}
				}

			}
			else_if = {
				limit = {
					check_variable = {
						which = uma_ec2_num_fallen_empires_middle
						value >= 0.32
					}
					check_variable = {
						which = uma_ec2_num_fallen_empires_middle
						value < 0.66
					}
				}

				every_playable_country = {
					limit = {
						has_event_chain = uma_ec_2_main_evt_chain
					}
					add_event_chain_counter = {
						event_chain = uma_ec_2_main_evt_chain
						counter = uma_ec2_fallen_empire_class
						amount = -1
					}
				}

				event_target:uma_fallen_empire = {
					remove_modifier = uma_fallen_empire_crisis_2_modifier_class_1_hull
					remove_modifier = uma_fallen_empire_crisis_2_modifier_class_1_speed
					remove_modifier = uma_fallen_empire_crisis_2_modifier_class_2_hull
					remove_modifier = uma_fallen_empire_crisis_2_modifier_class_2_speed
					remove_modifier = uma_fallen_empire_crisis_2_modifier_class_3_hull
					remove_modifier = uma_fallen_empire_crisis_2_modifier_class_4

					add_modifier = {
						modifier = uma_fallen_empire_crisis_2_modifier_class_2_speed
					}
					add_modifier = {
						modifier = uma_fallen_empire_crisis_2_modifier_class_2_hull
					}
				}
			}
			else_if = {
				limit = {
					check_variable = {
						which = uma_ec2_num_fallen_empires_middle
						value >= 0.65
					}
					check_variable = {
						which = uma_ec2_num_fallen_empires_middle
						value < 0.99
					}
				}
				set_global_flag = uma_ec_2_disable_scan_andkill_from_fe_capitial

				every_playable_country = {
					limit = {
						has_event_chain = uma_ec_2_main_evt_chain
					}
					add_event_chain_counter = {
						event_chain = uma_ec_2_main_evt_chain
						counter = uma_ec2_fallen_empire_class
						amount = -2
					}
				}

				event_target:uma_fallen_empire = {
					remove_modifier = uma_fallen_empire_crisis_2_modifier_class_1_hull
					remove_modifier = uma_fallen_empire_crisis_2_modifier_class_1_speed
					remove_modifier = uma_fallen_empire_crisis_2_modifier_class_2_hull
					remove_modifier = uma_fallen_empire_crisis_2_modifier_class_2_speed
					remove_modifier = uma_fallen_empire_crisis_2_modifier_class_3_hull
					remove_modifier = uma_fallen_empire_crisis_2_modifier_class_4
