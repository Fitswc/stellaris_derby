# Crisis Initiated
country_event = {
	id = uma_ec_3.1
	hide_window = yes
	fire_only_once = yes
	is_triggered_only = yes

	immediate = {
		set_global_flag = uma_ec3_outeruma_happened
		endgame_telemetry = uma_outer
		set_crisis_stage_1 = yes
		random_system = {
			limit = {
				has_astral_rift = no
				can_spawn_astral_rift = yes
			}
			save_global_event_target_as = uma_ec3_begining_astral_rift_system
			set_star_flag = uma_ec3_begining_astral_rift_system
			
			spawn_astral_rift = {
				random_pos = yes
				id = uma_ar_ec3
				graphics_entity_name = astral_rift_entity
				init_effect = {
					save_event_target_as = uma_ec3_ar
				}
			}

		}

		#An event that notice all country about the beginning of the crisis
		every_playable_country = {
			limit = {
				is_ai = no
			}
			country_event = { id = uma_ec_3.2 days = 10 }
			country_event = { id = uma_ec_3.6 days = 150 }
		}
	}
}
